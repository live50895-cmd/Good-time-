function compressImageFile(file,maxDim,quality,cb){
 const reader=new FileReader();
 reader.onload=ev=>{
  const img=new Image();
  img.onload=()=>{
   const scale=Math.min(1,maxDim/Math.max(img.width,img.height));
   const w=Math.max(1,Math.round(img.width*scale)),h=Math.max(1,Math.round(img.height*scale));
   const canvas=document.createElement('canvas');
   canvas.width=w;canvas.height=h;
   canvas.getContext('2d').drawImage(img,0,0,w,h);
   cb(canvas.toDataURL('image/jpeg',quality));
  };
  img.onerror=()=>cb(ev.target.result);
  img.src=ev.target.result;
 };
 reader.readAsDataURL(file);
}

 let pool=v;
 if(currentLang==='ur'){
  const urIn=v.filter(x=>/ur|en-IN|hi-IN|en-PK/i.test(x.lang));
  if(urIn.length)pool=urIn;
  else{const enM=v.filter(x=>/^en/i.test(x.lang));if(enM.length)pool=enM;}
 }else{
  const langMatch=v.filter(x=>x.lang.toLowerCase()===tl.toLowerCase()||x.lang.toLowerCase().startsWith(px));
  if(langMatch.length)pool=langMatch;
  else{const enM=v.filter(x=>/^en/i.test(x.lang));if(enM.length)pool=enM;}

function speakText(raw){

 lastSpokenText=text;
 unlockAudioForMobile();
 clearTimeout(voiceWatchdog); clearInterval(speechKeepAlive); voiceQueue=[]; window.__uttQueue=[];
 const el=document.getElementById('ttsAudioEl'); if(el)try{el.pause();el.onended=null;el.onerror=null;}catch(e){}

 const mode=document.getElementById('voiceModeSelect').value;
 const rate=parseFloat(document.getElementById('voiceSpeedSelect').value)||0.96;
 const gender=document.getElementById('voiceGenderSelect')?.value||'female';
 const pitch=gender==='female'?1.24:(gender==='male'?0.72:1.0);
 const gTag=gender==='female'?'👩 Girl':(gender==='male'?'👨 Man':'🤖 Auto');

 if(mode!=='online'){
  try{
   if(window.AndroidTTS?.speak){setVoiceUI(true,`📱 Android (${gTag})`,text);window.AndroidTTS.speak(text);return;}
   if(window.Android?.speak){setVoiceUI(true,`📱 Android (${gTag})`,text);window.Android.speak(text);return;}

  }catch(e){}
 }

 if(mode==='online'){playOnlineAudioTTS(text,rate);return;}

  try{if(window.speechSynthesis.speaking||window.speechSynthesis.pending)window.speechSynthesis.cancel();}catch(e){}
  const tl=langSpeechMap[currentLang]||'en-US';
  const bv=pickBestVoice(tl,gender);
  const chunks=splitIntoSpeechChunks(text,155);
  let idx=0;
  clearInterval(speechKeepAlive);
  speechKeepAlive=setInterval(()=>{
   if(window.speechSynthesis.speaking&&!window.speechSynthesis.paused){
    window.speechSynthesis.pause();window.speechSynthesis.resume();
   }
  },8500);

  const next=()=>{
   if(idx>=chunks.length){clearInterval(speechKeepAlive);setVoiceUI(false,'Finished — Ready',text);return;}
   const chText=chunks[idx];
   const u=new SpeechSynthesisUtterance(chText);
   window.__uttQueue.push(u);
   if(bv){u.voice=bv;u.lang=bv.lang;}else u.lang=tl;
   u.rate=rate; u.pitch=pitch; u.volume=1.0;

   u.onend=()=>{
    const elapsed=tStart?(performance.now()-tStart):999;
    if(idx===0&&chText.length>12&&elapsed<135&&mode==='auto'){
     clearTimeout(voiceWatchdog);clearInterval(speechKeepAlive);
     playOnlineAudioTTS(text,rate);
     return;
    }
    idx++;next();
   };
   u.onerror=()=>{
    clearTimeout(voiceWatchdog);clearInterval(speechKeepAlive);
    if(!started&&mode==='auto')playOnlineAudioTTS(text,rate);
    else setVoiceUI(true,'🌐 Switching to Online Audio...',text,true);
   };
   if(idx===0&&mode==='auto'){
    clearTimeout(voiceWatchdog);
    voiceWatchdog=setTimeout(()=>{
     if(!started){
      try{window.speechSynthesis.cancel();}catch(e){}
      clearInterval(speechKeepAlive);
      playOnlineAudioTTS(text,rate);
     }
    },950);
   }
   try{
    if(window.speechSynthesis.paused)window.speechSynthesis.resume();
    window.speechSynthesis.speak(u);
   }catch(e){
    clearTimeout(voiceWatchdog);clearInterval(speechKeepAlive);
    if(mode==='auto')playOnlineAudioTTS(text,rate);
   }
  };
  setVoiceUI(true,`🔊 Starting ${gTag} (${tl})...`,text);
  next();
 }else{
  playOnlineAudioTTS(text,rate);
 }
}
